    const headerContainer = document.getElementById("header-container");
    
    if (headerContainer) {
        fetch("C:\Users\Jude\Desktop\CSC 106\Header.html")
            .then(response => response.text())

            })
            .catch(error => console.error("Error loading header:", error));
    }
});
